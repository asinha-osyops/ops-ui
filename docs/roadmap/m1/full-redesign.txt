 import * as z from "zod"

 export const sopSchema = z.object({
   companyId: z.string().uuid("Please select a company"),
   sopName: z.string().min(1, "Required").max(255),
   basicDescription: z.string().min(1, "Required").max(1000),

 6.3 React Hook Form Pattern

 Old Pattern (useState):
 const [formData, setFormData] = useState({ ... })
 const handleSubmit = () => { ... }

 New Pattern (React Hook Form):
 import { useForm } from "react-hook-form"
 import { zodResolver } from "@hookform/resolvers/zod"
 import { Form, FormField, FormItem, FormLabel, FormControl, FormMessage } from "@/components/ui/form"

 const form = useForm({
   resolver: zodResolver(sopSchema),
   defaultValues: { sopName: "", ... }
 })

 const onSubmit = async (data) => {
   await apiClient.createSOP(data)
 }

 return (
   <Form {...form}>
     <form onSubmit={form.handleSubmit(onSubmit)}>
       <FormField
         control={form.control}
         name="sopName"
         render={({ field }) => (
           <FormItem>
             <FormLabel>SOP Name</FormLabel>
             <FormControl>
               <Input {...field} />
             </FormControl>
             <FormMessage />
           </FormItem>
         )}
       />
     </form>
   </Form>
 )
